\SpeedTitleBlock{Daily Geometry Drill \#5}{Henry Koduthore}
\SpeedMeta{Tangents from an external point}{tangent length $\sqrt{OP^2-r^2}$, angle between tangents $\sin\frac{\theta}{2}=\frac{r}{OP}$, chord of contact (polar) $xx_1+yy_1=r^2$, director circle $x^2+y^2=2r^2$}
\noindent\textit{New toolkit today: the polar of an external point $\cdot$ two equal tangents $\cdot$ $\sin\frac{\theta}{2}=\frac{r}{OP}$ $\cdot$ the director circle.}

\vspace{2pt}
\noindent\textit{Attempt each question before checking answers. Time yourself. No calculators.}

\section*{Section A \quad Rapid Recognition \hfill{\normalfont\small($\leq$15 s each)}}
% ══════════════════════════════════════════════════════════════════════════════

\noindent\textit{Read $OP$ and $r$ straight off, then the tangent length is one step of Pythagoras.}

\begin{enumerate}

\item Write down the length of the tangent from $(0,5)$ to $x^2+y^2=9$.

\item The point $(0,5)$ sits outside $x^2+y^2=9$ with $OP=5$ and $r=3$. Write down $\tan\frac{\theta}{2}$, where $\theta$ is the angle between the two tangents.

\item Write down the equation of the chord of contact (polar) drawn from the point $(5,3)$ to $x^2+y^2=25$.

\item The circle $x^2+y^2=25$ has director circle $x^2+y^2=2r^2$. Write down the radius of that director circle.

\item Write down the length of the tangent from $(7,0)$ to $x^2+y^2=36$.

\item Two tangents from the point $P$ to a circle of unknown radius make an angle of $60^\circ$ and $OP=6$. Write down the length of each tangent.

\item Write down the equation of the chord of contact drawn from $(5,0)$ to $x^2+y^2=4$ (the two tangency points join in a vertical line).

\item A tangent from the point $P$ to $x^2+y^2=25$ has length $12$. Write down $OP$.

\item The two tangents from $P$ to the circle $x^2+y^2=25$ make an angle of $120^\circ$. Write down $OP$.

\item The chord of contact from the point $P$ to $x^2+y^2=16$ is the line $x+y=4$. Write down $P$.

\end{enumerate}

\section*{Section B \quad Manipulation Drills \hfill{\normalfont\small($\leq$50 s each)}}
% ══════════════════════════════════════════════════════════════════════════════

\noindent\textit{Feed external points and radii through $\sqrt{OP^2-r^2}$ and the polar formula, choosing from the options.}

\begin{enumerate}

\item The length of the tangent from $(3,4)$ to $x^2+y^2=9$ is?
\begin{itemize}
\item[A)] $\sqrt{7}$
\item[B)] $4$
\item[C)] $5$
\item[D)] $2$
\end{itemize}

\item The point $P$ has $OP=6$ with respect to a circle of radius $3$. The angle between the two tangents from $P$ is?
\begin{itemize}
\item[A)] $30^\circ$
\item[B)] $45^\circ$
\item[C)] $60^\circ$
\item[D)] $90^\circ$
\end{itemize}

\item Write down the equation of the chord of contact drawn from the point $(3,5)$ to $x^2+y^2=25$.

\item The director circle of $x^2+y^2=4$ has radius?
\begin{itemize}
\item[A)] $2$
\item[B)] $2\sqrt{2}$
\item[C)] $4$
\item[D)] $\sqrt{2}$
\end{itemize}

\item The length of the tangent from $(1,2)$ to $x^2+y^2=1$ is?
\begin{itemize}
\item[A)] $1$
\item[B)] $\sqrt{5}$
\item[C)] $2$
\item[D)] $\sqrt{3}$
\end{itemize}

\item A tangent from $P$ to a circle of radius $5$ has length $12$. How far is $P$ from the centre?
\begin{itemize}
\item[A)] $17$
\item[B)] $7$
\item[C)] $13$
\item[D)] $\sqrt{119}$
\end{itemize}

\item The chord of contact from $P=(a,0)$ to $x^2+y^2=16$ is the line $x=4$. Find $a$.
\begin{itemize}
\item[A)] $2$
\item[B)] $4$
\item[C)] $8$
\item[D)] $16$
\end{itemize}

\item The two tangents from $P$ to $x^2+y^2=25$ meet at right angles (so $P$ lies on the director circle). Write down the length of each tangent.

\item The tangent to $x^2+y^2=8$ with slope $1$ has positive $y$-intercept equal to?
\begin{itemize}
\item[A)] $4$
\item[B)] $2\sqrt{2}$
\item[C)] $8$
\item[D)] $4\sqrt{2}$
\end{itemize}

\item A tangent from $P$ to a circle has length $6$, and $OP=10$. Write down the circle's radius.

\end{enumerate}

\section*{Section C \quad Substitution \& Structure \hfill{\normalfont\small($\leq$75 s each)}}
% ══════════════════════════════════════════════════════════════════════════════

\noindent\textit{Hard: equal tangents, director circle, and a hidden $s-a$ trap. No direct plug.}

\begin{enumerate}

\item From $P(8,12)$, two tangents are drawn to $x^2+y^2=36$ (centre $O$, $r=6$). Let $T_1T_2$ be the chord of contact. What is the distance from $O$ to the line $T_1T_2$? \textit{\small(after TMUA 2020 Paper 1 Q16 / BMO1 2013 Q4)}
\begin{itemize}
\item[A)] $\tfrac{9}{5}$
\item[B)] $\tfrac{18}{5}$
\item[C)] $\tfrac{36}{5}$
\item[D)] $\tfrac{9}{2}$
\end{itemize}

\item The locus of points $P$ where the tangent length to $x^2+y^2=4$ equals the tangent length to $(x-6)^2+y^2=16$ is the radical axis. Which line is it? \textit{\small(adapted from TMUA 2018 Paper 1 Q7)}
\begin{itemize}
\item[A)] $x=1$
\item[B)] $x=2$
\item[C)] $x=3$
\item[D)] $x=4$
\end{itemize}

\item \begin{tikzpicture}[scale=0.42, baseline=-0.5ex]
\draw (0,0) circle (1.2);
\draw (1.2,0) -- (2.5,0.8);
\draw[fill=black] (0,0) circle (1pt) node[below]{\tiny $O$};
\draw (0,0) -- (1.2,0) node[midway, below]{\tiny $r$};
\end{tikzpicture} The circle $x^2+y^2-6x-8y+24=0$ has centre $(3,4)$, $r=1$. The point $P(6,8)$ is outside. What is the length of the tangent from $P$ to the circle? \textit{\small(after MAT 2021 Q5)}
\begin{itemize}
\item[A)] $4$
\item[B)] $2\sqrt5$
\item[C)] $5$
\item[D)] $\sqrt{26}$
\end{itemize}

\item For the $13$-$14$-$15$ triangle ($s=21$, $r=4$), the two tangent lengths from the vertex opposite side $14$ are $s-b$. With $b=14$, what is that length? \textit{\small(adapted from BMO1 2015 Q3)}
\begin{itemize}
\item[A)] $6$
\item[B)] $7$
\item[C)] $8$
\item[D)] $9$
\end{itemize}

\item The circle $(x-4)^2+y^2=9$ has director circle $x^2+y^2=2r^2$ for a centred circle, but this circle is translated. What is the locus of points from which the two tangents to $(x-4)^2+y^2=9$ are perpendicular? \textit{\small(after TMUA 2022 Paper 1 Q19)}
\begin{itemize}
\item[A)] $(x-4)^2+y^2=18$
\item[B)] $(x-4)^2+y^2=9$
\item[C)] $x^2+y^2=18$
\item[D)] $x^2+y^2=9$
\end{itemize}

\item \begin{tikzpicture}[scale=0.4, baseline=-0.5ex]
\draw (0,0) circle (1.2);
\draw (-0.8,-0.8) -- (0.8,0.8);
\draw[fill=black] (0,0) circle (1pt);
\end{tikzpicture} The point $(8,6)$ is at distance $10$ from the origin. Tangents from $(8,6)$ to $x^2+y^2=1$ have length $\sqrt{99}$. What is the distance from the origin to the chord of contact? \textit{\small(adapted from TMUA 2020 Paper 1 Q16)}
\begin{itemize}
\item[A)] $\tfrac{1}{10}$
\item[B)] $\tfrac{1}{\sqrt{101}}$
\item[C)] $\tfrac{1}{\sqrt{99}}$
\item[D)] $\tfrac{10}{\sqrt{101}}$
\end{itemize}

\item The triangle with sides $13,14,15$ has incircle touching $BC$ at $D$. If $BD=6$, $DC=9$, and $s=21$, the two equal tangents from $A$ are $s-a$. What is $AE$ (tangents from $A$)? \textit{\small(after BMO1 2020 Q3)}
\begin{itemize}
\item[A)] $6$
\item[B)] $7$
\item[C)] $8$
\item[D)] $9$
\end{itemize}

\item The chord of contact of $(2,1)$ w.r.t. $x^2+y^2=1$ is $2x+y=1$. What is its distance from the origin, and is it inside, on, or outside the circle? \textit{\small(adapted from TMUA 2021 Paper 2 Q7)}
\begin{itemize}
\item[A)] $\tfrac{1}{\sqrt5}$ (inside, $d<r$)
\item[B)] $\tfrac{1}{\sqrt5}$ (outside, $d>r$)
\item[C)] $\tfrac{\sqrt5}{5}$ (inside)
\item[D)] $1$ (tangent)
\end{itemize}

\end{enumerate}

\section*{Section D \quad Challenge Ramp \hfill{\normalfont\small($\leq$90 s each)}}
% ══════════════════════════════════════════════════════════════════════════════

\noindent\textit{Hardest 5: angle $\theta/2$ via $r/OP$, chord $AB$ via power of a point, and a hidden incircle.}

\begin{enumerate}

\item \begin{tikzpicture}[scale=0.45, baseline=-0.5ex]
\draw (0,0) circle (1.2);
\draw (0,0) -- (1.5,0.8);
\draw (1.5,0.8) -- (0.8,1.5);
\node at (0.6,0.6) {\tiny $\theta$};
\draw[fill=black] (0,0) circle (1pt);
\end{tikzpicture} Two tangents from $P$ to $x^2+y^2=9$ with $OP=6$ enclose angle $\theta$. What is $\theta$? \textit{\small(after TMUA 2018 Paper 1 Q7 / BMO1 2023 Q4)}
\begin{itemize}
\item[A)] $30^\circ$
\item[B)] $60^\circ$
\item[C)] $90^\circ$
\item[D)] $120^\circ$
\end{itemize}

\item From $P$ to $x^2+y^2=25$, tangent length is $5\sqrt3$. Find $OP$ (use $OP^2=r^2+\text{tangent}^2$). \textit{\small(adapted from TMUA 2020 Paper 1 Q16)}
\begin{itemize}
\item[A)] $10$
\item[B)] $5\sqrt{10}$
\item[C)] $10\sqrt{3}$
\item[D)] $75$
\end{itemize}

\item The chord of contact from $P(3,3)$ to $x^2+y^2=9$ is $x+y=3$. What is its length? (Hint: distance from $O$ to chord is $3/\sqrt2$, $r=3$) \textit{\small(adapted from TMUA 2019 Paper 1 Q6)}
\begin{itemize}
\item[A)] $3\sqrt2$
\item[B)] $\tfrac{3\sqrt{14}}{2}$
\item[C)] $3\sqrt{6}$
\item[D)] $6$
\end{itemize}

\item Two tangents from $P$ make $60^\circ$ and $OP=10$, $r=OP\sin(\theta/2)=5$, tangent length $=OP\cos(\theta/2)$. What is the length? \textit{\small(after BMO1 2015 Q3)}
\begin{itemize}
\item[A)] $5$
\item[B)] $5\sqrt3$
\item[C)] $10$
\item[D)] $5/2$
\end{itemize}

\item The polar of $(9,12)$ w.r.t. $x^2+y^2=81$ is $3x+4y=27$. What is its distance from the origin, and is $P$ outside/inside/on? \textit{\small(adapted from MAT 2021 Q5)}
\begin{itemize}
\item[A)] $27/5$ (outside, $d<r$? No, $d=27/5=5.4<r=9$? Actually $r=9$, $d=5.4<r$, so chord inside)
\item[B)] $27/15$ (outside)
\item[C)] $9/5$ (inside)
\item[D)] $81/15$ (outside)
\end{itemize}

\end{enumerate}
