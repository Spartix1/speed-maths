\section*{Section A \quad Rapid Recognition}

\begin{enumerate}

\item \ans{"a circle of radius 5 centred at the origin"}
\method{$\sqrt{x^2+y^2}=5$ rearranges to $x^2+y^2=25$, a circle of radius $5$.}
\inv{Forgetting to square: quoting radius $25$ or a square.}

\item \ans{$x=2$}
\method{Squaring $x^2+y^2=(x-4)^2+y^2$: $x^2=(x-4)^2$, so $x=2$.}
\inv{Giving the single midpoint $(2,0)$ instead of the whole vertical line.}

\item \ans{$4$}
\method{On the $x$-axis, between $A$ and $B$ we have $PA=x$ and $PB=6-x$, so $x=2(6-x)$, giving $x=4$. The Apollonius circle's two $x$-axis endpoints are $x=4$ and $x=12$.}
\inv{Choosing $x=2$ (halving $6$) or the external endpoint $x=12$.}

\item \ans{$y=1$ or $y=7$}
\method{$|y-4|=3$ gives $y=1$ and $y=7$, two parallel lines $3$ units from $y=4$.}
\inv{Keeping only one line $y=1$ and missing its mirror $y=7$.}

\item \ans{$(2,1)$}
\method{A point equidistant from focus $(0,1)$ and directrix $y=-1$ satisfies $\sqrt{x^2+(y-1)^2}=|y+1|$, i.e.\ $x^2=4y$; $(2,1)$ works since $4=4$.}
\inv{Naming a point on the $y$-axis (which only hits the vertex origin for integer $y$) or mislaying the focus/directrix distance.}

\item \ans{$(x-1)^2+(y-2)^2=9$}
\method{Points at distance exactly $3$ from $(1,2)$ form the circle centred there with radius $3$.}
\inv{Writing distance $=3$ unsquared, or using the point as an offset in the wrong sign.}

\item \ans{"a circle of radius 1 centred at the origin"}
\method{$Q$ runs over $x^2+y^2=4$ (radius $2$); the midpoint $M=OQ/2$ scales radii by $\frac12$, giving $x^2+y^2=1$.}
\inv{Halving the equation's radius linearly to $\sqrt{2}$ or keeping the original radius $2$.}

\item \ans{$1$}
\method{$x^2+y^2=2x$ rearranges to $(x-1)^2+y^2=1$, radius $1$.}
\inv{Answering $\sqrt{2}$ (reading the constant $2$ as $r^2$ before completing the square).}

\item \ans{false}
\method{The point $\left(\sqrt2,\sqrt2\right)$ lies on $x^2+y^2=4$ yet has $x\ne\pm2$ and $y\ne\pm2$; the implication fails in both coordinates.}
\inv{Mistaking the circle's bounding box for its points; only the four axis points hit $\pm2$.}

\item \ans{$x=3$}
\method{$A$ and $B$ are at equal height $y=3$, so the perpendicular bisector is the vertical line $x=\frac{1+5}{2}=3$.}
\inv{Writing the horizontal line $y=3$ through both points.}

\end{enumerate}

\section*{Section B \quad Manipulation Drills}

\begin{enumerate}

\item \ans{A) $4$}
\method{On the $x$-axis, $|x|=2|x-6|$ gives the endpoints $x=4$ and $x=12$; the circle between them has centre $(8,0)$ and radius $4$.}
\inv{Quoting the centre distance $8$ (option B) or the gap $6$ (option D).}

\item \ans{B) $y=2$}
\method{$x^2+y^2=x^2+(y-4)^2$ gives $y=2$, the horizontal bisector.}
\inv{Answering $x=2$ (wrong orientation) or $y=4$.}

\item \ans{B) $4\pi$}
\method{The disk has radius $2$, so its area is $\pi r^2=4\pi$.}
\inv{Using $\pi r=2\pi$ (option A) or $r^2=16$ (option D).}

\item \ans{$2\pi$}
\method{Circumference $=2\pi r=2\pi\cdot 1=2\pi$.}
\inv{Mixing circumference with area ($\pi$) or doubling the radius to $4\pi$.}

\item \ans{C) $x^2+y^2<2$}
\method{If $x^2+y^2<1$ then certainly $x^2+y^2<2$; option A is not necessary (a point with $x^2+y^2=0.75$ is inside but not $<\frac12$).}
\inv{Picking a circle whose radius is smaller than the disk (option A) as if it were implied.}

\item \ans{A) $3$}
\method{On the $x$-axis $|x|=3|x-2|$ gives internal $x=\frac32$ and external $x=3$; the right endpoint is $3$.}
\inv{Choosing the left endpoint $\frac32$ or the $y$-absent centre guess.}

\item \ans{B) $x=-\dfrac12$ and $y=\dfrac{\sqrt3}{2}$}
\method{This point satisfies $x^2+y^2=\frac14+\frac34=1$ exactly, so it pins $P$ on the circle. Option A puts $P$ strictly {\em inside}; C and D do not force $x^2+y^2=1$.}
\inv{Choosing a strictly-inside condition (option A) because it is a {\em necessary} symptom of being on the circle.}

\item \ans{$3$}
\method{Distance from $(3,0)$ to the origin is $\sqrt{9}=3$.}
\inv{Answering $6$ (the segment $AB$) or $9$ (the squared distance).}

\item \ans{$(2,3)$}
\method{In $x^2+y^2+2gx+2fy+c=0$, the centre is $(-g,-f)$; with $2g=-4$ and $2f=-6$ we get $(2,3)$.}
\inv{Halving the wrong sign: $(4,6)$ or negating incorrectly.}

\item \ans{$2$}
\method{Completing the square: $(x-2)^2+(y+3)^2=(-9)+4+9=4$, radius $2$.}
\inv{Quoting the constant $9$ as the radius or $r^2$ before shifting.}

\end{enumerate}

\section*{Section C \quad Substitution \& Structure}

\begin{enumerate}

%── C1 ──────────────────────────────────────────────────────────────────────────
\item \ans{A) $(6,0)$}
\method{Square $PA=2PB$: $(x+2)^2+y^2=4\big((x-4)^2+y^2\big)$. Expand: $x^2+4x+4+y^2=4x^2-32x+64+4y^2$, so $3x^2-36x+60+3y^2=0$, i.e.\ $x^2-12x+20+y^2=0$, $(x-6)^2+y^2=16$. Centre $(6,0)$, radius $4$. Option A. The $x$-axis endpoints from $|x+2|=2|x-4|$ are $x=2$ (internal) and $x=10$, midpoint $6$.}
\inv{Quoting the external endpoint $x=10$ or the midpoint of $AB$, $(1,0)$ (options C/D).}

%── C2 ──────────────────────────────────────────────────────────────────────────
\item \ans{D) $x=0$ or $y=0$}
\method{Distance to $L_1$ is $|x+y|/\sqrt2$, to $L_2$ is $|x-y|/\sqrt2$. Equate: $|x+y|=|x-y|$. Squaring: $(x+y)^2=(x-y)^2$, so $4xy=0$, i.e.\ $x=0$ or $y=0$, the two coordinate axes (the internal and external angle bisectors of $y=\pm x$). Options A/B give only one bisector; C is not a locus of points equidistant from lines.}
\inv{Answering a single line $y=x$ (option A) which is the original $L_2$ itself, not its bisector; forgetting the modulus gives the second branch.}

%── C3 ──────────────────────────────────────────────────────────────────────────
\item \ans{A)}
\method{Fil invokes the internal angle-bisector theorem which requires $D$ to be {\em interior} to $AB$. The diagram may place $D$ on the extension of $AB$ beyond $B$ (or beyond $A$) where the external bisector ratio is $AD/DB=-CA/CB$; without proving convexity/betweenness the betweenness hypothesis is not established. This is the classic ``all triangles are isosceles'' fallacy class. Hence the argument assumes what it must prove. Option A names the betweenness gap; B miscasts a squared modulus, C misstates the theorem, D claims no flaw.}
\inv{Misidentifying the gap as a sign/square issue (option B) or declaring the flawed proof sound (option D).}

%── C4 ──────────────────────────────────────────────────────────────────────────
\item \ans{E) none of them individually (all insufficient alone)}
\method{For a parallelogram: I.\ $PQ=QR$ forces a rhombus; a rhombus with angles $60^\circ,120^\circ$ (e.g.\ equilateral-parallelogram) is not a square. II.\ $PR\perp QS$ also forces a rhombus (equivalently to I) but not a square (same counterexample). III.\ $\angle PQR=90^\circ$ forces a rectangle; a $2\times3$ rectangle is not a square. So each alone is insufficient; $I+III$ or $II+III$ together would be sufficient. Hence no single condition suffices, option E. This mirrors TMUA 2020 P2 Q7 answer ``none individually sufficient'' (official answer H in the 8-option paper, here condensed to E).}
\inv{Choosing I or III alone (options A/C) because ``rhombus'' or ``right angle'' sounds like ``square'' without checking the other degree of freedom.}

%── C5 ──────────────────────────────────────────────────────────────────────────
\item \ans{E) $p=2$ or $p\ge4$}
\method{By the sine rule $\sin R / PR = \sin 30^\circ / p$, so $\sin R = 2/p$. The ambiguous case: for $p<4$ there are potentially two values of $R$ giving two triangles, except when $\sin R=1$ ($p=2$) where the two coalesce to a right-angled triangle uniquely determined. For $p\ge PR=4$ the side opposite the given angle is at least the other given side, so the angle $R$ is uniquely acute; $p=2$ also unique. Hence exactly $p=2$ or $p\ge4$ gives a unique $PQ$; all other $p<4$, $p\ne2$ give two possible $PQ$ (or none for $p<2\sin30^\circ$). This scales the official TMUA 2018 P2 Q14 answer $p=1$ or $p\ge2$ (with $PR=2$) by factor $2$.}
\inv{Answering $p\ge4$ alone (option D) and missing the isolated right-angle case $p=2$; or quoting the unscaled $p=1$ threshold (option A).}

%── C6 ──────────────────────────────────────────────────────────────────────────
\item \ans{E) $4$}
\method{Centres $O_1(-3,1)$, $O_2(8,1)$, so $d=O_1O_2=11$. Radii $r_1=5$, $r_2=3$. Since $d=11>r_1+r_2=8$ the circles are externally separate: $4$ common tangents (2 direct, 2 transverse). Compare $d$ with $r_1+r_2$ and $|r_1-r_2|=2$: $d>r_1+r_2\Rightarrow4$, $d=r_1+r_2\Rightarrow3$, $|r_1-r_2|<d<r_1+r_2\Rightarrow2$, $d=|r_1-r_2|\Rightarrow1$, $d<|r_1-r_2|\Rightarrow0$. Here $11>8$ so $4$.}
\inv{Counting $3$ (option D) by mistaking $d=r_1+r_2$ tangency, or $2$ by thinking intersecting.}

%── C7 ──────────────────────────────────────────────────────────────────────────
\item \ans{A) $9$}
\method{For $PA=3PB$, $x^2+y^2=9\big((x-6)^2+y^2\big)$ gives $8x^2-108x+8y^2+324=0$, i.e.\ $(x-\tfrac{27}{4})^2+y^2=(\tfrac94)^2$. Centre $C(\tfrac{27}{4},0)= (6.75,0)$, radius $r=\tfrac94=2.25$. The furthest point from the origin on the circle is on the ray $OC$ beyond $C$: $OC+r=6.75+2.25=9$; the nearest is $OC-r=4.5$. So $\max OP=9$, option A.}
\inv{Quoting $OC=6.75$ (not a listed option) or the nearest distance $4.5$/diameter $4.5$.}

%── C8 ──────────────────────────────────────────────────────────────────────────
\item \ans{A)}
\method{Expand correctly: $x^2+y^2=9\big((x-2)^2+y^2\big)$ gives $x^2+y^2=9x^2-36x+36+9y^2$, so $8x^2-36x+8y^2+36=0$, i.e.\ $(x-\tfrac94)^2+y^2=\tfrac{9}{16}$, centre $(\tfrac94,0)$, radius $\tfrac34$. Fil's $(x-2)^2+y^2=9$ is the circle centred at $B=(2,0)$ with radius $3$; it uses only $PB$ and ignores $A$ entirely. Option A's locus is the true Apollonius circle; B claims the locus collapses, C claims it is a line ($k=1$ only), D claims no error.}
\inv{Accepting the $B$-centred circle (option D) because its centre ``looks like $B$'' and its radius ``is the ratio times $1$''.}

\end{enumerate}

\section*{Section D \quad Challenge Ramp}

\begin{enumerate}

%── D1 ──────────────────────────────────────────────────────────────────────────
\item \ans{A) $2\sqrt2$}
\method{$PA^2=(x+3)^2+y^2$, $PB^2=(x-3)^2+y^2$. Sum $=2x^2+18+2y^2=34$, so $x^2+y^2=8$, i.e.\ $(x-0)^2+y^2=8$, a circle centred at the midpoint $(0,0)$ with $r^2=8$, $r=2\sqrt2$. This is the $PA^2+PB^2$ locus (Apollonius's theorem generalisation). Check: $PA^2+PB^2=2m^2+AB^2/2$ with $m=OP$. Option A; B is $r^2$, D is total sum.}
\inv{Answering $\sqrt8$ as $8$ (option B/D confusion) or forgetting the factor $2$ before completing the square.}

%── D2 ──────────────────────────────────────────────────────────────────────────
\item \ans{C) $18$}
\method{$|x|+|y|=3$ is four lines $x+y=3$, $x-y=3$, etc., forming a diamond with vertices $(3,0),(0,3),(-3,0),(0,-3)$. Its diagonals are $6$ and $6$, so area $=\frac12\cdot6\cdot6=18$ (equivalently four right triangles each $\tfrac12\cdot3\cdot3=4.5$). For $|x|+|y|=1$ the area is $2$ (TMUA 2021 P1 Q9 answer C), scaling by factor $3^2$ gives $2\cdot9=18$. Option C.}
\inv{Doubling instead of squaring the scale factor (giving $12$/$6$, options A/B).}

%── D3 ──────────────────────────────────────────────────────────────────────────
\item \ans{C) $x^2=yz$}
\method{Put $O$ at origin, let $SP^2=x^2+z^2$, $PQ^2=x^2+y^2$, $SQ^2=(y+z)^2$. Then $\triangle SPQ$ right at $P$ iff $SP^2+PQ^2=SQ^2$ (Pythagoras): $(x^2+z^2)+(x^2+y^2)=(y+z)^2$, so $2x^2+y^2+z^2=y^2+2yz+z^2$, i.e.\ $2x^2=2yz$, $x^2=yz$. Conversely $x^2=yz$ reverses the steps; hence the condition is both necessary and sufficient. This is exactly TMUA 2022 P2 Q11 (official answer C). Option C.}
\inv{Choosing $y=z$ (option D) which only makes the kite symmetric but still an isosceles right possibility, not the product condition.}

%── D4 ──────────────────────────────────────────────────────────────────────────
\item \ans{B) $6$}
\method{Radical axis: subtract $x^2+y^2=25$ and $(x-4)^2+y^2=9$, i.e.\ $x^2-(x-4)^2=16$, so $8x-16=16$, $x=4$. Distance from $O_1(0,0)$ to the line $x=4$ is $4$. Half-chord $=\sqrt{r_1^2-d_{\rm axis}^2}=\sqrt{25-16}=3$, so full chord $PQ=6$. Check with $O_2$: $4-4=0$, distance $0$, half-chord also $\sqrt{9-0}=3$. Option B. The diagram's dashed line is $x=4$.}
\inv{Quoting the half-chord $3$ (option A) or using $d=O_1O_2=4$ directly as the chord.}

%── D5 ──────────────────────────────────────────────────────────────────────────
\item \ans{$12$}
\method{Apollonius $PA=2PB$ with $A(0,0),B(6,0)$ is $(x-8)^2+y^2=16$, centre $C(8,0)$, radius $r=4$. For fixed base $AB=6$, $[PAB]=\tfrac12\cdot AB\cdot|y_P|$, so maximise $|y_P|$ among points on the circle: $\max|y_P|=r=4$ (topmost point $(8,4)$). Hence $\max[PAB]=\tfrac12\cdot6\cdot4=12$. Check: $(8,4)$ indeed satisfies $(0,4)$? $PA=\sqrt{80}=4\sqrt5$, $PB=\sqrt{20}=2\sqrt5$, ratio $2$ correct. Uniqueness up to reflection in $x$-axis.}
\inv{Using $r=16$ (option squared) or $AB=12$ (diameter) for $24$.}

\end{enumerate}

\section*{Top 5 Patterns}

\begin{itemize}
\item \textbf{Apollonius ratio $PA/PB=k\neq1$ squares to a circle; read its centre by squaring and completing the square.} The $x$-axis endpoints from $|x-a|=k|x-b|$ give the diameter; the midpoint is the centre. \seealso{C1, C7, C8, D5}
\item \textbf{$PA^2+PB^2$ and $|x|+|y|$ loci are also locus families: sum-of-squares is a centred circle; $|x|+|y|=a$ is a diamond whose area scales as $a^2$.} Use $AB$ midpoint and the diagonal product. \seealso{D1, D2}
\item \textbf{Distinguish {\em necessary} from {\em sufficient} and spot the hidden betweenness hypothesis.} A condition is necessary if every square has it; sufficient if it alone forces a square; a proof that needs ``$D$ between $A$ and $B$'' must prove convexity first. \seealso{C3, C4, D3}
\item \textbf{The ambiguous case (sine rule) decides uniqueness via $\sin R = PR\sin P/p$; the isolated $\sin R=1$ case and the $p\ge PR$ regime are the only uniquely-determined ones.} \seealso{C5}
\item \textbf{Subtract circles for the radical axis; its distance from the centre gives the common chord via $\sqrt{r^2-d_{\rm axis}^2}$.} External $d>r_1+r_2$ counts four tangents. \seealso{C6, D4}
\end{itemize}

\section*{Common Traps}

\begin{itemize}
\item \textbf{Writing an Apollonius locus centred at $B$ (or at the midpoint of $AB$) instead of completing the squared ratio.} The true centre divides $AB$ externally/internally in ratio $k^2\!:\!1$, not $1\!:\!1$. \seealso{C1, C7, C8, D5}
\item \textbf{Collapsing a modulus locus to one branch.} $|x+y|=|x-y|$ has {\em two} solutions $x=0$ and $y=0$; $|y-4|=3$ has twins $y=1,7$; $|PA|=k|PB|$ has internal {\em and} external $x$-axis endpoints. \seealso{A4, C2, C7}
\item \textbf{Claiming a rhombus or right-angle condition alone forces a square.} Each degree of freedom (side equality vs right angle) must be secured separately. \seealso{C4, D3}
\item \textbf{Forgetting the scale factor for diamond/bounded loci.} Area($|x|+|y|=a$) $=2a^2$; doubling $a$ quadruples area, not doubles. \seealso{D2}
\item \textbf{Quoting a half-chord or a squared radius as the full length.} The distance from the centre to the radical axis is not the chord; $\sqrt{r^2-d^2}$ is the half-chord, $r^2$ is not $r$. \seealso{D1, D4}
\end{itemize}
