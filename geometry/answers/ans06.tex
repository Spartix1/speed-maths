\section*{Section A \quad Rapid Recognition}

\begin{enumerate}

\item \ans{$40$}
\method{The angle at the centre is twice the angle at the circumference on the same arc: $2\times 20^\circ = 40^\circ$.}
\inv{Halving instead of doubling; the centre angle is the bigger one.}

\item \ans{$35$}
\method{$\angle PQR$ and $\angle PSR$ stand on the same chord $PR$, so they are equal.}
\inv{Watching the wrong chord; both angles must look at $PR$.}

\item \ans{$110$}
\method{Opposite angles of a cyclic quadrilateral sum to $180^\circ$: $\angle C = 180^\circ - 70^\circ = 110^\circ$.}
\inv{Using $\angle B$ (adjacent, not opposite) instead of $\angle A$.}

\item \ans{$40$}
\method{The tangent--chord angle equals the angle in the alternate segment subtended by that chord, so it is $40^\circ$.}
\inv{Quoting the angle's supplement $140^\circ$; the equal angle is on the far side of the chord.}

\item \ans{$4$}
\method{Intersecting chords: $PA\cdot PB=PC\cdot PD$, so $2\cdot 6 = 3\cdot PD$, $PD=4$.}
\inv{Equating $PA=PD$; the products, not the pairs, are equal.}

\item \ans{$9$}
\method{Tangent--secant power (Day 5): $PT^2 = PA\cdot PB$, so $9 = 1\cdot PB$, $PB=9$.}
\inv{Forgetting that $PAB$ is $P$--$A$--$B$: $PB$ is the full secant, not the chord $AB=8$.}

\item \ans{$90$}
\method{Thales: a diameter subtends a right angle at the circumference.}
\inv{Answering $180^\circ$ (the straight diameter line) rather than the right angle it subtends.}

\item \ans{$12$}
\method{Secant--secant power: $PA\cdot PB = PC\cdot PD$, so $4\cdot 6 = 2\cdot PD$, $PD=12$.}
\inv{Exchanging a near point for a far point; $PB=6$ is the {\em full} $P$--$A$--$B$ segment.}

\item \ans{$16$}
\method{$PT^2 = PA\cdot PB$: $64=4\cdot PB$, $PB=16$.}
\inv{Quoting the chord $AB = PB-PA = 12$ instead of the full $PB$.}

\item \ans{$30$}
\method{$\angle BAC$ and $\angle BDC$ stand on the same chord $BC$, so both are $30^\circ$.}
\inv{Adding the diagonal's split angle; only the chord $BC$ matters.}

\end{enumerate}

\section*{Section B \quad Manipulation Drills}

\begin{enumerate}

\item \ans{B) $65$}
\method{Half the central angle: $\frac12\cdot 130^\circ = 65^\circ$.}
\inv{Answering the centre angle itself ($130^\circ$, option D).}

\item \ans{A) $80$}
\method{Opposite angles are supplementary: $180^\circ - 100^\circ = 80^\circ$.}
\inv{Keeping the angle as-is because the quadrilateral looks symmetric.}

\item \ans{A) $1$}
\method{$3\cdot 5 = 15\cdot PD$, so $PD=1$.}
\inv{Multiplying $3$ by $5$ and dividing by the nearer point $5$, getting a mirror of the given $PC$.}

\item \ans{$\dfrac{25}{2}$}
\method{$PT^2=PA\cdot PB$: $25 = 2\cdot PB$, $PB=\frac{25}{2}$.}
\inv{Answering $5$ (the tangent half) or dividing $25$ by $PB$ instead of $PA$.}

\item \ans{B) $55$}
\method{Alternate segment theorem: the tangent--chord angle equals the angle in the alternate segment.}
\inv{Taking $90^\circ-55^\circ=35^\circ$ (a tangent--radius reflex).}

\item \ans{A) $6$}
\method{$PT=\sqrt{PA\cdot PB}=\sqrt{2\cdot 18}=\sqrt{36}=6$.}
\inv{Adding segments into $\frac{PA+PB}{2}=10$ rather than taking the geometric mean.}

\item \ans{C) $36$}
\method{Opposite angles sum to $180^\circ$: $2x+3x=5x=180^\circ$, $x=36$.}
\inv{Setting $2x=3x$ or forgetting the opposite pairing.}

\item \ans{$35$}
\method{$\angle ACB=90^\circ$ (diameter), so $\angle BAC = 180^\circ - 90^\circ - 55^\circ = 35^\circ$.}
\inv{Answering $55^\circ$ as if the triangle were isosceles.}

\item \ans{A) $4$}
\method{Pitot: $AB+CD=BC+DA$, so $DA=5+6-7=4$.}
\inv{Pairing adjacent sides instead of opposite ones.}

\item \ans{$8$}
\method{$PA\cdot PB=PC\cdot PD$: $6\cdot 4=3\cdot PD$, $PD=8$.}
\inv{Treating $PB$ as a single piece rather than $P$--$A$--$B$ ordering.}

\end{enumerate}

\section*{Section C \quad Substitution \& Structure}

\begin{enumerate}

%── C1 ──────────────────────────────────────────────────────────────────────────
\item \ans{B) $6$}
\method{Let $CP=3k$, $PD=4k$. Intersecting chords: $PA\cdot PB=CP\cdot PD$, i.e.\ $6\cdot 8=3k\cdot 4k=12k^{2}$. So $48=12k^{2}$, $k^{2}=4$, $k=2$, $CP=6$ and $PD=8$. Check: $6$ and $8$ are consecutive multiples of $3,4$.}
\inv{Dividing $48$ by $7$ ($3+4$) instead of $12$ ($3\cdot4$); the power is a product, not a sum.}

%── C2 ──────────────────────────────────────────────────────────────────────────
\item \ans{B) $10$}
\method{Tangent--secant power: $PT^{2}=PA\cdot PB$. So $144=8\cdot PB$, $PB=18$, hence chord $AB=PB-PA=10$. The diagram is a right triangle $OPT$ with hypotenuse $OP$.}
\inv{Quoting $PB=18$ as the chord, or subtracting the wrong segment ($PT-PA$).}

%── C3 ──────────────────────────────────────────────────────────────────────────
\item \ans{C) $6$}
\method{Ptolemy for cyclic $ABCD$: $AC\cdot BD=AB\cdot CD+BC\cdot AD =6\cdot5+4\cdot3=30+12=42$. With $AC=7$, $BD=42/7=6$. Verify $BD<AB+AD$ etc.\ so cyclic existence holds.}
\inv{Multiplying adjacent sides ($6\cdot4+5\cdot3$) instead of opposite pairs, or quoting the product $42$ as a diagonal.}

%── C4 ──────────────────────────────────────────────────────────────────────────
\item \ans{A) $10\sqrt{6}$}
\method{$s=\tfrac{4+5+5+6}{2}=10$. Brahmagupta: $K=\sqrt{(10-4)(10-5)(10-5)(10-6)}=\sqrt{6\cdot5\cdot5\cdot4}=\sqrt{600}=10\sqrt6$.}
\inv{Using perimeter $20$ for $s$, or applying Heron to a triangle after an illegal diagonal split.}

%── C5 ──────────────────────────────────────────────────────────────────────────
\item \ans{B) $15$}
\method{Pitot for tangential quadrilaterals: $AB+CD=BC+DA$. So $3x+5=8+2x+3$, giving $3x+5=2x+11$, $x=6$, hence $DA=2x+3=15$. Check opposite sums: $18+5=23$, $8+15=23$.}
\inv{Pairing adjacent sides ($AB+BC=CD+DA$) or solving $3x=2x$ and dropping constants.}

%── C6 ──────────────────────────────────────────────────────────────────────────
\item \ans{A) $6$}
\method{Secant--secant power from $P$: $PA\cdot PB=PC\cdot PD$. $PA\cdot PB=4\cdot18=72$, so $6\cdot PD=72$, $PD=12$, chord $CD=PD-PC=6$. Same product $72$ is $6\times12$.}
\inv{Using $PA$ as $PB$ (near vs far confusion) or quoting $PD=12$ as the chord.}

%── C7 ──────────────────────────────────────────────────────────────────────────
\item \ans{A) $7$}
\method{Ptolemy: $AC\cdot BD=AB\cdot CD+BC\cdot DA=8\cdot6+6\cdot6=48+36=84$. With $AC=12$, $BD=84/12=7$. Check $7<8+6$ etc.\ valid.}
\inv{Adding adjacent products ($AB\cdot BC+CD\cdot DA$) or dividing the sum by the wrong diagonal.}

%── C8 ──────────────────────────────────────────────────────────────────────────
\item \ans{A) $44$}
\method{$s=\tfrac{5+5+8+14}{2}=16$. Brahmagupta: $K=\sqrt{(16-5)(16-5)(16-8)(16-14)}$\\ $=\sqrt{11\cdot11\cdot8\cdot2}=\sqrt{1936}=44$.}
\inv{Using $s=32$ (full perimeter) or factoring $\sqrt{1936}$ as $22\sqrt4$ and miscounting.}

\end{enumerate}

\section*{Section D \quad Challenge Ramp}

\begin{enumerate}

%── D1 ──────────────────────────────────────────────────────────────────────────
\item \ans{A) $36$}
\method{$s=\tfrac{2+5+10+11}{2}=14$. $K=\sqrt{(14-2)(14-5)(14-10)(14-11)}=\sqrt{12\cdot9\cdot4\cdot3}=\sqrt{1296}=36$. Check trapezoid? Also equals Heron split via diagonal $12$: two triangles $5\!-\!12\!-\!13$ style. }
\inv{Forgetting to halve the perimeter, or leaving $\sqrt{1296}$ as a surd.}

%── D2 ──────────────────────────────────────────────────────────────────────────
\item \ans{A) $11$}
\method{Ptolemy: $AC\cdot BD=AB\cdot CD+BC\cdot DA=7\cdot7+8\cdot9=49+72=121$. With $AC=11$, $BD=121/11=11$. Both diagonals equal $11$; quadrilateral is an isosceles trapezoid in disguise.}
\inv{Multiplying adjacent sides ($7\cdot8+7\cdot9$) or quoting the product $121$ as a diagonal.}

%── D3 ──────────────────────────────────────────────────────────────────────────
\item \ans{B) $8$}
\method{Tangent--secant power: $PT^{2}=PA\cdot PB$ with $PB=PA+10$. So $144=PA(PA+10)$, i.e.\ $PA^{2}+10PA-144=0$, $(PA+18)(PA-8)=0$, $PA=8$. Check: $8\cdot18=144=12^{2}$.}
\inv{Solving $PA^{2}+10PA=144$ as $PA=12$ or $PA=14$ by linear division, or forgetting $PB$ is $PA+10$ not $10$.}

%── D4 ──────────────────────────────────────────────────────────────────────────
\item \ans{A) $40$}
\method{Sides $5,5,8,8$ (order $AB=5,BC=5,CD=8,DA=8$): Pitot check $AB+CD=5+8=13=BC+DA$, so tangential confirms an incircle exists. Semiperimeter $s=13$. Brahmagupta (cyclic): $K=\sqrt{(13-5)(13-5)(13-8)(13-8)}=\sqrt{8\cdot8\cdot5\cdot5}=40$. Bicentric: both theorems hold; area also $=r\cdot s$ with $r=40/13$.}
\inv{Using Heron on a split triangle, or pairing Pitot incorrectly and claiming tangential fails.}

%── D5 ──────────────────────────────────────────────────────────────────────────
\item \ans{A) $19$}
\method{Ptolemy: $AC\cdot BD=AB\cdot CD+BC\cdot DA=6\cdot12+8\cdot10=72+80=152$. With $AC=8$, $BD=152/8=19$. Check $19<6+8+10$ etc.\ valid diagonal. }
\inv{Swapping a near-diagonal for a side, or quoting $152$ (the sum) as a diagonal.}

\end{enumerate}

\section*{Top 5 Patterns}

\begin{itemize}
\item \textbf{Product power: $PA\cdot PB=PC\cdot PD$ and $PT^{2}=PA\cdot PB$.} Chords crossing, tangent--secant, or two secants from one exterior point all reduce to one product equation. \seealso{A5, A6, A8, A9, B3, B4, B6, B10, C1, C2, C6, D3}
\item \textbf{Ptolemy adds opposite products.} $AC\cdot BD = AB\cdot CD + BC\cdot AD$ for any cyclic quadrilateral; when one diagonal and all sides are known the other diagonal is one division. \seealso{C3, C7, D2, D5}
\item \textbf{Brahmagupta is Heron with two extra sides.} $K=\sqrt{(s-a)(s-b)(s-c)(s-d)}$ with $s=\tfrac{a+b+c+d}{2}$; the product under the root must be a perfect square for clean integers. \seealso{C4, C8, D1, D4}
\item \textbf{Pitot pairs opposite sides of a tangential quadrilateral.} $AB+CD=BC+DA$ is necessary and sufficient for an incircle; algebraic sides turn this into a linear equation. \seealso{B9, C5, D4}
\item \textbf{Cyclic opposites are supplementary and same-segment angles are equal.} $\angle A+\angle C=180^\circ$ and angles standing on the same chord are equal -- the angle-chase that justifies ``cyclic'' claims. \seealso{A3, A10, B2, B7, C3}
\end{itemize}

\section*{Common Traps}

\begin{itemize}
\item \textbf{Quoting the full secant instead of the chord.} $AB=PB-PA$, not $PB$ itself; tangent--secant always asks for $AB$ you must subtract. \seealso{A6, A9, C2, C6, D3}
\item \textbf{Multiplying adjacent sides in Ptolemy.} The products are {\em opposite} pairs $AB\cdot CD$ and $BC\cdot AD$, never $AB\cdot BC$. \seealso{C3, C7, D2, D5}
\item \textbf{Using the perimeter for $s$ in Brahmagupta.} Semiperimeter is half the perimeter ($s=\tfrac{P}{2}$); doubling it squares the error. \seealso{C4, C8, D1, D4}
\item \textbf{Pairing Pitot on adjacent sides.} Pitot is $AB+CD=BC+DA$ (opposites), not $AB+BC=CD+DA$. \seealso{B9, C5, D4}
\item \textbf{Dividing power products by a sum, not a product.} Intersecting chords: $CP\cdot PD=PA\cdot PB$; with $CP:PD=3:4$ set $CP=3k$, $PD=4k$ so $12k^{2}=48$, not $(3+4)k$. \seealso{A5, C1, C6, B10}
\end{itemize}
